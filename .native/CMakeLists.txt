cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(dst_fastloading_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(LIBDEFLATE_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(LIBDEFLATE_BUILD_GZIP OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/libdeflate)
add_subdirectory(vendor/minhook)
add_library(dst_fastloading_native SHARED src/plugin.cpp src/zip_reader.cpp)
target_link_libraries(dst_fastloading_native PRIVATE libdeflate_static minhook)
target_compile_options(dst_fastloading_native PRIVATE /W4 /utf-8)
enable_testing()
add_executable(test_zip tests/test_zip.cpp src/zip_reader.cpp)
target_include_directories(test_zip PRIVATE src)
target_link_libraries(test_zip PRIVATE libdeflate_static)
target_compile_options(test_zip PRIVATE /W4 /utf-8)
add_test(NAME zip_reader COMMAND test_zip)
set(SMOOTH_LOADING_SOURCE "" CACHE PATH "Optional smooth_loading source for cross-mod hook tests")
if(SMOOTH_LOADING_SOURCE)
    add_executable(test_zip_hook tests/test_zip_hook.cpp src/zip_reader.cpp)
    target_include_directories(test_zip_hook PRIVATE src "${SMOOTH_LOADING_SOURCE}/.native/src")
    target_link_libraries(test_zip_hook PRIVATE minhook libdeflate_static)
    target_compile_options(test_zip_hook PRIVATE /W4 /utf-8)
    add_test(NAME zip_hook COMMAND test_zip_hook)
endif()
